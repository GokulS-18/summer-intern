<script>
const $ = id => document.getElementById(id);
let ingredients = [];

/* ---------- ingredient chips ---------- */
const QUICK = ["Eggs","Tomato","Onion","Rice","Chicken","Potato","Cheese","Garlic","Paneer","Bread"];
QUICK.forEach(q => { const b = document.createElement("button"); b.type="button"; b.textContent = "+ " + q; b.onclick = () => addIng(q); $("quick").appendChild(b); });
function renderChips(){
  $("chipBox").querySelectorAll(".chip").forEach(c => c.remove());
  ingredients.forEach((ing,i) => {
    const c = document.createElement("span"); c.className = "chip"; c.textContent = ing;
    const x = document.createElement("button"); x.type="button"; x.textContent="×"; x.setAttribute("aria-label","Remove "+ing);
    x.onclick = e => { e.stopPropagation(); ingredients.splice(i,1); renderChips(); };
    c.appendChild(x); $("chipBox").insertBefore(c, $("ingInput"));
  });
}
function addIng(v){
  v = v.trim(); if(!v) return;
  if(!ingredients.some(i => i.toLowerCase()===v.toLowerCase())) ingredients.push(v);
  $("ingInput").value=""; renderChips();
}
$("chipBox").onclick = () => $("ingInput").focus();
$("ingInput").addEventListener("keydown", e => {
  if(e.key==="Enter" || e.key===","){ e.preventDefault(); addIng($("ingInput").value); }
  else if(e.key==="Backspace" && !$("ingInput").value){ ingredients.pop(); renderChips(); }
});

/* ---------- API helper ---------- */
async function api(path, body){
  const r = await fetch(path, {method:"POST", headers:{"Content-Type":"application/json"}, body: JSON.stringify(body||{})});
  let data = {}; try { data = await r.json(); } catch(e){}
  if(!r.ok) throw new Error(data.error || "Something went wrong. Please try again.");
  return data;
}

/* ---------- generate recipe ---------- */
$("goBtn").onclick = async () => {
  addIng($("ingInput").value);
  $("setupError").classList.add("hidden");
  if(!ingredients.length){ showSetupError("Please add at least one ingredient."); return; }
  $("goBtn").disabled = true; $("goBtn").innerHTML = '<span class="spinner"></span> Cooking up a recipe…';
  try{
    const d = await api("/api/start", {ingredients: ingredients.join(", ")});
    $("setup").classList.add("hidden"); $("recipeCard").classList.remove("hidden"); $("chatCard").classList.remove("hidden");
    $("chat").innerHTML = ""; $("suggest").innerHTML = "";
    $("rName").textContent = d.name;
    $("rIngredients").innerHTML = ""; (d.ingredients||[]).forEach(t => { const s=document.createElement("span"); s.textContent=t; $("rIngredients").appendChild(s); });
    renderSteps(d);
    addMsg("ai", "Hi! I'm your chef. Cook along and press “Done” after each step. Ask me about substitutions, timing or tips anytime.");
    ["Can I substitute an ingredient?","How do I know it's cooked?","Any tips for more flavour?"].forEach(t => { const b=document.createElement("button"); b.type="button"; b.textContent=t; b.onclick=()=>{ $("chatInput").value=t; $("chatForm").requestSubmit(); }; $("suggest").appendChild(b); });
    window.scrollTo({top:0,behavior:"smooth"});
  }catch(e){ showSetupError(e.message); }
  finally{ $("goBtn").disabled=false; $("goBtn").textContent="Generate recipe"; }
};
function showSetupError(m){ $("setupError").textContent=m; $("setupError").classList.remove("hidden"); }

/* ---------- steps ---------- */
function renderSteps(d){
  const list = $("steps"); const had = list.children.length; list.innerHTML = "";
  d.steps.forEach((t,i) => {
    const done = i+1 < d.current || d.finished;
    const li = document.createElement("li");
    li.className = "step " + (done ? "done" : "current");
    li.innerHTML = '<div class="num"></div><div><strong></strong><p></p></div>';
    li.querySelector(".num").textContent = done ? "✓" : i+1;
    li.querySelector("strong").textContent = "Step " + (i+1);
    li.querySelector("p").textContent = t;
    list.appendChild(li);
  });
  const doneCount = d.finished ? d.total : d.current-1;
  $("bar").style.width = (doneCount/d.total*100)+"%";
  $("barLabel").textContent = d.finished ? "All "+d.total+" steps complete" : "Step "+d.current+" of "+d.total;
  $("finished").classList.toggle("hidden", !d.finished);
  $("nextBtn").classList.toggle("hidden", d.finished);
  if(had && list.lastElementChild) list.lastElementChild.scrollIntoView({behavior:"smooth",block:"center"});
}
$("nextBtn").onclick = async () => {
  $("nextBtn").disabled = true;
  try{ renderSteps(await api("/api/next")); }catch(e){ addMsg("ai","⚠️ "+e.message); }
  finally{ $("nextBtn").disabled=false; }
};

/* ---------- chat ---------- */
function addMsg(who, text){
  const m = document.createElement("div"); m.className = "msg "+who; m.textContent = text;
  $("chat").appendChild(m); $("chat").scrollTop = $("chat").scrollHeight; return m;
}
$("chatForm").onsubmit = async e => {
  e.preventDefault();
  const text = $("chatInput").value.trim(); if(!text) return;
  $("chatInput").value = ""; addMsg("me", text); $("sendBtn").disabled = true;
  const wait = addMsg("ai",""); wait.innerHTML = '<span class="typing"><i></i><i></i><i></i></span>';
  try{
    const d = await api("/api/chat", {message:text});
    if(d.action==="next"){ wait.textContent = d.finished ? "Wonderful, you finished the recipe! 🎉" : "Great! Here is the next step 👇"; renderSteps(d); }
    else wait.textContent = d.reply;
  }catch(err){ wait.textContent = "⚠️ " + err.message; }
  finally{ $("sendBtn").disabled=false; $("chatInput").focus(); $("chat").scrollTop=$("chat").scrollHeight; }
};

/* ---------- new recipe ---------- */
$("newBtn").onclick = async () => {
  try{ await api("/api/reset"); }catch(e){}
  ingredients = []; renderChips();
  $("recipeCard").classList.add("hidden"); $("chatCard").classList.add("hidden"); $("setup").classList.remove("hidden");
  window.scrollTo({top:0,behavior:"smooth"}); $("ingInput").focus();
};
</script>
